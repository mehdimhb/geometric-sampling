def bootstrap_interval(values,seed,stat='mean'):
    values=np.asarray(values,dtype=float); values=values[np.isfinite(values)]
    if not len(values): return np.nan,np.nan
    rng=np.random.default_rng(seed)
    draws=rng.choice(values,(1000,len(values)),replace=True)
    samples=np.mean(draws,axis=1) if stat=='mean' else np.median(draws,axis=1)
    return tuple(np.quantile(samples,[.025,.975]))


def summarize(raw,master):
    rows=[]
    for keys,g in raw.groupby(GROUP,dropna=False,sort=True):
        row=dict(zip(GROUP,keys)); row['runs']=len(g); row['valid_results']=int(g.final_target_RMSE2.notna().sum())
        row['status_counts']=json.dumps(g.status.value_counts().to_dict(),sort_keys=True)
        feasible=g[g.feasible==True]
        row['feasible_runs']=len(feasible)
        for label in ['recovery_1e4','recovery_1e6','recovery_1e8']:
            count=int(feasible[label].sum()); denom=len(feasible)
            rate=count/denom if denom else np.nan
            row[label+'_rate']=rate
            # Wilson interval, including failures among known-feasible targets.
            if denom:
                z=1.959963984540054; center=(rate+z*z/(2*denom))/(1+z*z/denom)
                half=z*np.sqrt(rate*(1-rate)/denom+z*z/(4*denom*denom))/(1+z*z/denom)
                row[label+'_lower']=center-half; row[label+'_upper']=center+half
        for metric in METRICS:
            v=g[metric].dropna().to_numpy(float)
            if not len(v): continue
            row[metric+'_mean']=float(np.mean(v)); row[metric+'_median']=float(np.median(v))
            row[metric+'_SD']=float(np.std(v,ddof=1)) if len(v)>1 else np.nan
            q1,q3=np.quantile(v,[.25,.75]); row[metric+'_Q1']=q1; row[metric+'_Q3']=q3
            row[metric+'_IQR']=q3-q1; row[metric+'_min']=float(np.min(v)); row[metric+'_max']=float(np.max(v))
            for stat in ['mean','median']:
                lo,hi=bootstrap_interval(v,seed_for(master,str(keys),metric,stat),stat)
                row[metric+'_'+stat+'_CI_lower']=lo; row[metric+'_'+stat+'_CI_upper']=hi
        rows.append(row)
    return pd.DataFrame(rows)


def paired(raw,master):
    keys=['scenario_id','replication','regime','evaluation_budget','time_budget_seconds']
    primary=raw[raw.method=='Package-GFS'][keys+['final_target_RMSE2']].rename(columns={'final_target_RMSE2':'primary_RMSE2'})
    merged=raw[raw.method!='Package-GFS'].merge(primary,on=keys,how='inner')
    merged['delta_RMSE2']=merged.final_target_RMSE2-merged.primary_RMSE2
    group=['experiment','N','n','target_type','geometry','target_density','pair_structure','method','method_variant','regime']
    rows=[]
    for labels,g in merged.groupby(group,dropna=False):
        v=g.delta_RMSE2.dropna().to_numpy()
        row=dict(zip(group,labels)); row['complete_pairs']=len(v); row['attempted_pairs']=len(g)
        if len(v):
            row.update({'mean_competitor_minus_primary':v.mean(),'median_competitor_minus_primary':np.median(v)})
            lo,hi=bootstrap_interval(v,seed_for(master,str(labels)))
            row.update({'mean_difference_CI_lower':lo,'mean_difference_CI_upper':hi})
        rows.append(row)
    return pd.DataFrame(rows)


def save_figure(root,number,fig):
    fig.tight_layout()
    fig.savefig(root/f'figures/figure_{number}.pdf',bbox_inches='tight')
    fig.savefig(root/f'figures/figure_{number}.png',dpi=300,bbox_inches='tight')
    plt.close(fig)


def unavailable(ax,title):
    ax.set_title(title); ax.text(.5,.5,'No executed data for this experiment',ha='center',va='center',transform=ax.transAxes)
    ax.set_xticks([]); ax.set_yticks([])


def plot_curves(ax,checkpoints,x,scenario=None):
    data=checkpoints.copy()
    if scenario: data=data[data.scenario_id==scenario]
    if 'regime' in data: data=data[data.regime=='evaluations']
    data=data[(data.replication==0)&(data[x]>0)]
    if not len(data): unavailable(ax,'Convergence'); return
    groups=['method','method_variant'] if 'method_variant' in data else ['method']
    for keys,g in data.groupby(groups,sort=False):
        method=keys[0] if isinstance(keys,tuple) else keys
        variant=keys[1] if isinstance(keys,tuple) and len(keys)>1 else 'default'
        if not (method.startswith('GFS-') or method=='Package-GFS'): continue
        g=g.sort_values(x)
        # Random shows its working trajectory; optimization methods show incumbent.
        y='target_RMSE2' if method=='GFS-Random' else 'incumbent_target_RMSE2'
        ax.plot(g[x],np.maximum(g[y],1e-16),label=method+(f' ({variant})' if variant!='default' else ''),linewidth=1.3)
    ax.set_xscale('log'); ax.set_yscale('log'); ax.set_ylabel('Target-pair SIP RMSE')
    ax.legend(fontsize=7); ax.grid(alpha=.2)

def tables(root,summary):
    columns=['experiment','target_density','pair_structure','regime','evaluation_budget','time_budget_seconds','candidate_batch','N','n','geometry','target_type','FIP_type','method','method_variant','runs','final_target_RMSE2_median',
             'final_target_RMSE2_mean','final_Dinf_median','recovery_1e6_rate','runtime_seconds_median',
             'final_support_size_median','QP_optimal_D2_median','GFS_excess_over_QP_median','status_counts']
    for number,experiments in [(1,['small']),(2,['moderate']),(3,['large','sparse']),(4,['ablation'])]:
        table=summary[summary.experiment.isin(experiments)]
        table=table[[c for c in columns if c in table]]
        table.to_csv(root/f'tables/table_{number}.csv',index=False)
        table.to_latex(root/f'tables/table_{number}.tex',index=False,float_format=lambda x:f'{x:.3g}',escape=True,longtable=True)
    summary[summary.geometry=='1d'].to_csv(root/'tables/one_dimensional_summary.csv',index=False)
    summary[summary.geometry=='1d'][[c for c in columns if c in summary]].to_latex(root/'tables/one_dimensional_summary.tex',index=False,float_format=lambda x:f'{x:.3g}',escape=True,longtable=True)

def generate(root):
    root=Path(root); raw=pd.read_csv(root/'results/runs.csv')
    try: checkpoints=pd.read_csv(root/'results/checkpoints.csv')
    except pd.errors.EmptyDataError: checkpoints=pd.DataFrame(columns=['scenario_id','replication','candidate_moves','runtime_seconds','method'])
    manifest=json.loads((root/'results/manifest.json').read_text()); master=manifest['config']['master_seed']
    summary=summarize(raw,master); comparisons=paired(raw,master)
    summary.to_csv(root/'results/summary.csv',index=False); comparisons.to_csv(root/'results/paired_comparisons.csv',index=False)
    raw[~raw.status.isin(['completed','budget_exhausted','certified_feasible','certified_qp'])].to_csv(root/'results/failures.csv',index=False)
    tables(root,summary); representative=figures(root,raw,checkpoints); report(root,raw,summary,comparisons,representative)
